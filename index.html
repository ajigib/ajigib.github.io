def gc_content(seq):
    gc = seq.count("G") + seq.count("C")
    return gc / len(seq)

def find_hgt_candidates(genes, donors):
    hits = homology_search(genes, donors)
    for gene in genes:
        index = hgt_index(hits[gene])
        if index > THRESHOLD:
            yield gene, index

def enrich(gene_set, background):
    p = hypergeom_test(gene_set, background)
    return fdr_correct(p)

def synteny_blocks(anchors, gap=50_000):
    blocks, cur = [], [anchors[0]]
    for a in anchors[1:]:
        if a.pos - cur[-1].pos <= gap:
            cur.append(a)
        else:
            blocks.append(cur); cur = [a]
    return blocks + [cur]

Research

My work sits between evolutionary genomics and software engineering: finding where genomes have been rearranged or have borrowed genes, and packaging the methods so that others can reproduce them.

Comparative genomics

Multi-species genome analysis, synteny reconstruction and detection of evolutionary breakpoints.

Horizontal gene transfer

Detection using homology-based and composition-based approaches in multicellular organisms.

Genome annotation & evolution

Ancestral genome reconstruction, repeat and transposable element annotation, functional enrichment.
